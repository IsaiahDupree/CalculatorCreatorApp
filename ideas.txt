import type { TCalculatorSpec } from '@calc/spec/src/spec'

const math = create(all, {}) as MathJsStatic

export type EvalResult = { outputs: Record<string, number | string>; warnings: string[] }

function toScopeFromInputs(spec: TCalculatorSpec, inputs: Record<string, any>) {
  const scope: Record<string, any> = {}
  for (const inp of spec.inputs) {
    const v = inputs[inp.id]
    if (inp.unit) {
      scope[inp.id] = math.unit(v, inp.unit)
    } else {
      scope[inp.id] = v
    }
  }
  return scope
}
